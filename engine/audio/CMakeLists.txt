# dream::audio: playback through SDL3. Optional like the renderer: a build without SDL3 still
# writes sound to a WAV file through the launcher's --wav. A NextUI build (DREAM_RENDER_NEXTUI)
# plays through the handheld's SDL2 instead; SDL2 never goes into a PC port's executable, whose
# SDL3 exports the same symbol names.
find_package(SDL3 QUIET CONFIG)
if(SDL3_FOUND)
  add_library(dream_audio STATIC src/sdl_sink.cpp)
  target_link_libraries(dream_audio PUBLIC SDL3::SDL3)
  message(STATUS "dream::audio: SDL3 playback enabled")
elseif(DREAM_RENDER_NEXTUI)
  find_path(DREAM_AUDIO_SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
  find_library(DREAM_AUDIO_SDL2_LIBRARY SDL2)
  if(NOT DREAM_AUDIO_SDL2_INCLUDE_DIR OR NOT DREAM_AUDIO_SDL2_LIBRARY)
    message(STATUS "dream::audio: no SDL2 for the NextUI build; playback is skipped")
    return()
  endif()
  add_library(dream_audio STATIC src/sdl2_sink.cpp)
  target_include_directories(dream_audio PRIVATE ${DREAM_AUDIO_SDL2_INCLUDE_DIR})
  target_link_libraries(dream_audio PUBLIC ${DREAM_AUDIO_SDL2_LIBRARY})
  message(STATUS "dream::audio: SDL2 playback enabled (NextUI)")
else()
  message(STATUS "dream::audio: no SDL3; playback is skipped")
  return()
endif()
target_include_directories(dream_audio PUBLIC include)
dream_target_defaults(dream_audio)
add_library(dream::audio ALIAS dream_audio)
